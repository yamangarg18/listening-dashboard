## Dependencies (requirements.txt)

```
httpx>=0.27.0            # HTTP client for all API calls (async-capable, timeout support)
openai>=1.40.0           # Official OpenAI Python SDK
python-dotenv>=1.0.0     # Load .env files
apscheduler>=3.10.0      # Job scheduling (cron-like triggers)
tenacity>=8.2.0          # Retry logic with exponential backoff
```

### Why each dependency:
- **httpx**: Used for Notion API and Parallel API calls. Preferred over `requests` because it supports async, timeouts, and connection pooling natively.
- **openai**: Official SDK handles auth, retries, and structured output parsing. Do NOT use raw httpx for OpenAI calls.
- **python-dotenv**: Loads API keys from `.env` file so they're never hardcoded.
- **apscheduler**: Handles the 4 daily search cycles + 1 daily brief cycle. Supports cron expressions and timezone-aware scheduling.
- **tenacity**: Wraps API calls with retry logic (exponential backoff). Critical for handling Notion's 3 req/sec rate limit and transient API failures.

## config.py

```python
"""
Configuration loader.
Reads API keys from .env, database IDs from database_ids.json.
"""

import os
import json
from dotenv import load_dotenv

load_dotenv()


# --- API Keys ---
NOTION_API_KEY = os.getenv("NOTION_API_KEY")
PARALLEL_API_KEY = os.getenv("PARALLEL_API_KEY")
OPENAI_API_KEY = os.getenv("OPENAI_API_KEY")

# --- Notion ---
LISTENING_DASHBOARD_PAGE_ID = os.getenv("LISTENING_DASHBOARD_PAGE_ID")
NOTION_API_VERSION = "2022-06-28"
NOTION_BASE_URL = "https://api.notion.com/v1"

# --- Email ---
SMTP_HOST = os.getenv("SMTP_HOST", "smtp.gmail.com")
SMTP_PORT = int(os.getenv("SMTP_PORT", "587"))
SMTP_USER = os.getenv("SMTP_USER")
SMTP_PASSWORD = os.getenv("SMTP_PASSWORD")
ESCALATION_TO_EMAIL = os.getenv("ESCALATION_TO_EMAIL")

# --- Topic ---
TOPIC = os.getenv("TOPIC", "Physical AI")

# --- Database IDs (loaded after setup_notion.py runs) ---
DATABASE_IDS_FILE = "database_ids.json"

def load_database_ids() -> dict:
    """Load database IDs from the auto-generated JSON file."""
    if not os.path.exists(DATABASE_IDS_FILE):
        return {}
    with open(DATABASE_IDS_FILE, "r") as f:
        return json.load(f)

def save_database_ids(ids: dict) -> None:
    """Save database IDs to the JSON file."""
    with open(DATABASE_IDS_FILE, "w") as f:
        json.dump(ids, f, indent=2)

def get_database_id(name: str) -> str:
    """
    Get a specific database ID by name.
    Valid names: 'ingestion_queue', 'main_kb', 'synthesis', 'brief_log'
    Raises ValueError if database_ids.json doesn't exist or name not found.
    """
    ids = load_database_ids()
    if not ids:
        raise ValueError(
            "database_ids.json not found. Run setup_notion.py first."
        )
    if name not in ids:
        raise ValueError(
            f"Database '{name}' not found in database_ids.json. "
            f"Available: {list(ids.keys())}"
        )
    return ids[name]


# --- State tracking ---
STATE_FILE = "state.json"

def load_state() -> dict:
    """Load pipeline state (last run timestamps, etc.)."""
    if not os.path.exists(STATE_FILE):
        return {}
    with open(STATE_FILE, "r") as f:
        return json.load(f)

def save_state(state: dict) -> None:
    """Save pipeline state."""
    with open(STATE_FILE, "w") as f:
        json.dump(state, f, indent=2)


# --- Validation ---
def validate_config() -> list[str]:
    """
    Check that all required config values are set.
    Returns a list of error messages. Empty list = all good.
    """
    errors = []
    if not NOTION_API_KEY:
        errors.append("NOTION_API_KEY is not set in .env")
    if not LISTENING_DASHBOARD_PAGE_ID:
        errors.append("LISTENING_DASHBOARD_PAGE_ID is not set in .env")
    if not PARALLEL_API_KEY:
        errors.append("PARALLEL_API_KEY is not set in .env")
    if not OPENAI_API_KEY:
        errors.append("OPENAI_API_KEY is not set in .env")
    return errors
```

## logger.py

```python
"""
Logging configuration.
All modules import logger from here.
"""

import logging
import sys
from datetime import datetime


def setup_logger(name: str = "pipeline", level: int = logging.INFO) -> logging.Logger:
    """
    Create a logger that writes to both console and a daily log file.
    """
    logger = logging.getLogger(name)
    
    if logger.handlers:
        return logger  # Already configured
    
    logger.setLevel(level)
    formatter = logging.Formatter(
        "%(asctime)s | %(levelname)-7s | %(name)s | %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S"
    )

    # Console handler
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setFormatter(formatter)
    logger.addHandler(console_handler)

    # File handler (daily rotation by filename)
    log_filename = f"pipeline_{datetime.now().strftime('%Y%m%d')}.log"
    file_handler = logging.FileHandler(log_filename, encoding="utf-8")
    file_handler.setFormatter(formatter)
    logger.addHandler(file_handler)

    return logger
```

## main.py (Chunk 0 version — skeleton only)

```python
"""
Listening Pipeline — Entry Point
Run this file to start the pipeline.
"""

from config import validate_config, load_database_ids
from logger import setup_logger

logger = setup_logger()


def main():
    logger.info("=== Listening Pipeline Starting ===")
    
    # Step 1: Validate config
    errors = validate_config()
    if errors:
        for error in errors:
            logger.error(f"Config error: {error}")
        logger.error("Fix the above errors in .env and try again.")
        return
    
    logger.info("Config validated — all API keys present.")
    
    # Step 2: Check if databases are set up
    db_ids = load_database_ids()
    if not db_ids:
        logger.warning(
            "No database IDs found. Run setup_notion.py first to create "
            "the Notion databases."
        )
        return
    
    logger.info(f"Database IDs loaded: {list(db_ids.keys())}")
    
    # Step 3: Scheduler will be added in Chunk 16
    logger.info("Pipeline skeleton is ready. No scheduler configured yet.")
    logger.info("=== Listening Pipeline Stopped ===")


if __name__ == "__main__":
    main()
```

## Edge Cases Handled

- **Missing .env file**: `python-dotenv` silently returns None for all vars. `validate_config()` catches this and reports which keys are missing.
- **Missing database_ids.json**: `load_database_ids()` returns empty dict. `main.py` warns and exits gracefully.
- **Corrupted database_ids.json**: `json.load()` will raise `json.JSONDecodeError`. This is intentional — a corrupted state file should crash loudly, not silently continue.
- **Re-running main.py**: Safe to run multiple times. No side effects.
